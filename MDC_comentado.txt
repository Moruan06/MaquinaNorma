1  ZER C 3 2      ; Testa se C já está zerado. Se C = 0, vai para 3; senão, vai para 2.
2  SUB C 1        ; Decrementa C e volta para 1, repetindo até zerar C.

3  ZER D 5 4      ; Testa se D já está zerado. Se D = 0, vai para 5; senão, vai para 4.
4  SUB D 3        ; Decrementa D e volta para 3, repetindo até zerar D.

5  ZER B 22 6     ; Testa B. Se B = 0, o MDC está em A e o programa vai copiá-lo para C em 22.
6  ZER A 25 7     ; Testa A. Se A = 0, o MDC está em B e o programa vai copiá-lo para C em 25.

7  ZER B 12 8     ; Durante o cálculo de A mod B, testa B. Se B = 0, uma subtração completa foi feita e B deve ser restaurado em 12.
8  ZER A 15 9     ; Testa A. Se A = 0 antes de B, terminou a comparação e começa a reorganização dos valores em 15.
9  SUB A 10       ; Decrementa A em 1 como parte da subtração simultânea entre A e B.
10 SUB B 11       ; Decrementa B em 1 como parte da mesma subtração.
11 ADD D 7        ; Incrementa D para registrar quantas unidades foram retiradas de B e volta ao teste em 7.

12 ZER D 5 13     ; Testa D. Se D = 0, B já foi restaurado e volta ao início do algoritmo em 5.
13 SUB D 14       ; Decrementa D em 1 para retirar uma unidade do contador auxiliar.
14 ADD B 12       ; Incrementa B em 1, restaurando o valor original de B, e volta para 12.

15 ZER B 18 16    ; Transfere para A a parte restante de B. Se B = 0, segue para 18.
16 SUB B 17       ; Decrementa B em 1 durante a transferência.
17 ADD A 15       ; Incrementa A em 1 e volta para 15, concluindo a transferência de B para A.

18 ZER D 5 19     ; Testa D. Se D = 0, a reorganização terminou e volta ao início do algoritmo em 5.
19 SUB D 20       ; Decrementa D em 1.
20 ADD A 21       ; Incrementa A em 1 para completar em A o antigo valor de B.
21 ADD B 18       ; Incrementa B em 1 para guardar o resto A mod B e volta para 18.

22 ZER A 28 23    ; Caso B = 0, testa A. Se A = 0, a cópia terminou e o programa encerra em 28.
23 SUB A 24       ; Decrementa A em 1 durante a transferência do resultado.
24 ADD C 22       ; Incrementa C em 1, copiando o MDC de A para C, e volta para 22.

25 ZER B 28 26    ; Caso A = 0, testa B. Se B = 0, a cópia terminou e o programa encerra em 28.
26 SUB B 27       ; Decrementa B em 1 durante a transferência do resultado.
27 ADD C 25       ; Incrementa C em 1, copiando o MDC de B para C, e volta para 25.
