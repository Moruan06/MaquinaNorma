   // --- Cálculo do Resto da Divisão (D mod C) ---
   até D = 0

      se D = 0 então fim; // Divisão exata encontrada fora do subloop -> Não é primo (B=0)

      faça (

         D := D - 1;
         F := F - 1
      )
   );

   proximo_divisor:
   C := C + 1 // Incrementa o divisor se a rodada anterior falhou em dividir exatamente
