MDC de A e B usando C e D

C := 0;    // garante que C começa vazio
D := 0;    // garante que D começa vazio

até A = 0 ou B = 0
faça (
   
   // cálculo de A mod B por subtrações sucessivas

   até B = 0 ou A = 0
   faça (
      A := A - 1;
      B := B - 1;
      D := D + 1
   );

   se B = 0
   então (
      // restaura B usando D
      até D = 0
      faça (
         D := D - 1;
         B := B + 1
      )
   )

   senão (
      // A chegou a 0 antes de B
      // reorganiza os valores:
      // A recebe o antigo B
      // B recebe o resto da divisão

      até B = 0
      faça (
         B := B - 1;
         A := A + 1
      );

      até D = 0
      faça (
         D := D - 1;
         A := A + 1;
         B := B + 1
      )
   )
);

se B = 0
então (
   // o MDC está em A
   até A = 0
   faça (
      A := A - 1;
      C := C + 1
   )
)

senão (
   // o MDC está em B
   até B = 0
   faça (
      B := B - 1;
      C := C + 1
   )
)

1  ZER C 3 2      ; se C ≠ 0 então C := C - 1 até zerar C
2  SUB C 1        ; continua zerando C

3  ZER D 5 4      ; se D ≠ 0 então D := D - 1 até zerar D
4  SUB D 3        ; continua zerando D

5  ZER B 22 6     ; se B = 0 então o MDC está em A; senão continua
6  ZER A 25 7     ; se A = 0 então o MDC está em B; senão continua

7  ZER B 12 8     ; testa se a subtração simultânea terminou
8  ZER A 15 9     ; se A zerar antes de B inicia reorganização dos valores
9  SUB A 10       ; A := A - 1
10 SUB B 11       ; B := B - 1
11 ADD D 7        ; D := D + 1 usando D como contador auxiliar

12 ZER D 5 13     ; se D = 0 então B já foi restaurado
13 SUB D 14       ; D := D - 1
14 ADD B 12       ; B := B + 1 restaurando o divisor

15 ZER B 18 16    ; transfere o restante de B para A
16 SUB B 17       ; B := B - 1
17 ADD A 15       ; A := A + 1

18 ZER D 5 19     ; termina a reorganização quando D = 0
19 SUB D 20       ; D := D - 1
20 ADD A 21       ; A := A + 1
21 ADD B 18       ; B := B + 1 guardando o resto

22 ZER A 28 23    ; copia A para C quando B = 0
23 SUB A 24       ; A := A - 1
24 ADD C 22       ; C := C + 1

25 ZER B 28 26    ; copia B para C quando A = 0
